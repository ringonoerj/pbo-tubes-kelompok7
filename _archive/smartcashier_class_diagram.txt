```mermaid
classDiagram
    %% Inheritance and Abstract Class
    class User {
        <<abstract>>
        #String id
        #String name
        #String phoneNumber
        +displayProfile()* void
    }

    class Karyawan {
        -String employeeId
        -String password
        -double kpiScore
        +login(String inputId, String inputPassword) boolean
        +addKpi(double score) void
        +displayProfile() void
    }

    class Member {
        -String joinDate
        -double discountRate
        +applyDiscount(double subtotal) double
        +displayProfile() void
    }

    User <|-- Karyawan
    User <|-- Member

    %% Core Business Entities
    class Product {
        -Integer id
        -String productName
        -String category
        -Integer quantity
        -Double price
    }

    class Transaction {
        -Integer id
        -String transactionId
        -Karyawan karyawan
        -Member member
        -Double totalAmount
        -LocalDateTime transactionDate
        -List~TransactionItem~ items
        +calculateTotal() void
    }

    class TransactionItem {
        -Integer id
        -Transaction transaction
        -Product product
        -Integer quantity
        -Double subtotal
    }

    %% Entity Relationships
    Transaction "1" *-- "many" TransactionItem : contains
    Transaction "many" --> "1" Karyawan : handled by
    Transaction "many" --> "0..1" Member : discount for
    TransactionItem "many" --> "1" Product : reference

    %% Repositories (JPA Interfaces)
    class KaryawanRepository {
        <<interface>>
        +findByEmployeeId(String employeeId) Optional~Karyawan~
    }
    class MemberRepository {
        <<interface>>
        +findByPhoneNumber(String phoneNumber) Optional~Member~
    }
    class ProductRepository {
        <<interface>>
    }
    class TransactionRepository {
        <<interface>>
        +findByTransactionId(String transactionId) Optional~Transaction~
    }

    %% Services
    class CalculatorTools {
        <<utility>>
        +add(double a, double b)$ double
        +subtract(double a, double b)$ double
        +multiply(double a, double b)$ double
        +divide(double a, double b)$ double
        +modulo(double a, double b)$ double
    }

    class ProductService {
        +decreaseStock(Integer productId, int quantity) void
    }

    class TransactionService {
        -TransactionRepository transactionRepository
        -KaryawanRepository karyawanRepository
        -MemberRepository memberRepository
        -ProductRepository productRepository
        -ProductService productService
        +createTransaction(TransactionRequest request, String karyawanId) ReceiptResponse
        +getReceipt(String transactionId) ReceiptResponse
    }

    class KaryawanService {
        -KaryawanRepository karyawanRepository
    }

    class MemberService {
        -MemberRepository memberRepository
    }

    %% Service Dependencies
    TransactionService ..> CalculatorTools : uses
    TransactionService --> TransactionRepository : queries
    TransactionService --> KaryawanRepository : queries
    TransactionService --> MemberRepository : queries
    TransactionService --> ProductService : calls
    ProductService --> ProductRepository : updates
    KaryawanService --> KaryawanRepository : queries
    MemberService --> MemberRepository : queries

    %% Controllers
    class AuthController {
        -KaryawanRepository karyawanRepository
        +login(LoginRequest request) ResponseEntity
    }

    class ProductController {
        -ProductRepository productRepository
        +getAllProducts() List~Product~
        +getProductById(Integer id) ResponseEntity
        +createProduct(Product product) Product
        +updateProduct(Integer id, Product productDetails) ResponseEntity
        +deleteProduct(Integer id) ResponseEntity
    }

    class MemberController {
        -MemberRepository memberRepository
        +registerMember(Member member) Member
        +getMemberByPhone(String phone) ResponseEntity
    }

    class TransactionController {
        -TransactionService transactionService
        +checkout(TransactionRequest request, String karyawanId) ResponseEntity
        +getReceipt(String id) ResponseEntity
    }

    %% Controller to Service / Repository Associations
    AuthController --> KaryawanRepository
    ProductController --> ProductRepository
    MemberController --> MemberRepository
    TransactionController --> TransactionService

    %% DTOs (Data Transfer Objects)
    class LoginRequest {
        -String employeeId
        -String password
    }
    class LoginResponse {
        -boolean success
        -String id
        -String name
        -double kpiScore
    }
    class TransactionRequest {
        -String memberPhone
        -List~ItemRequest~ items
    }
    class ReceiptResponse {
        -String transactionId
        -String karyawanName
        -String memberName
        -double discountRate
        -Double totalAmount
        -LocalDateTime transactionDate
        -List~ReceiptItem~ items
    }

    %% DTO Usage
    AuthController ..> LoginRequest : accepts
    AuthController ..> LoginResponse : returns
    TransactionController ..> TransactionRequest : accepts
    TransactionController ..> ReceiptResponse : returns
    TransactionService ..> TransactionRequest : processes
    TransactionService ..> ReceiptResponse : creates
```
